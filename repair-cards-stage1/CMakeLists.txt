cmake_minimum_required(VERSION 3.20)
project(RepairCards VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(sqlite3 STATIC third_party/sqlite3.c)
target_include_directories(sqlite3 PUBLIC third_party)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_link_libraries(sqlite3 PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(repair_core src/store.cpp src/sequence.cpp)
target_include_directories(repair_core PUBLIC src third_party)
target_link_libraries(repair_core PUBLIC sqlite3)
add_executable(repair_server src/server.cpp)
target_link_libraries(repair_server PRIVATE repair_core Threads::Threads)
if(WIN32)
  target_compile_definitions(repair_server PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_server PRIVATE ws2_32)
endif()
if(MSVC)
  target_compile_options(repair_core PRIVATE /utf-8 /W4)
  target_compile_options(repair_server PRIVATE /utf-8 /W4)
else()
  target_compile_options(repair_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_compile_options(repair_server PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
endif()
add_custom_command(TARGET repair_server POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/templates" "$<TARGET_FILE_DIR:repair_server>/templates"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/config.example.json" "$<TARGET_FILE_DIR:repair_server>/config.example.json")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE repair_core)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core_tests COMMAND core_tests)
endif()
