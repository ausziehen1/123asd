cmake_minimum_required(VERSION 3.20)
project(RepairCards VERSION 0.4.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(sqlite3 STATIC third_party/sqlite3.c)
target_include_directories(sqlite3 PUBLIC third_party)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_link_libraries(sqlite3 PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(repair_core src/store.cpp src/sequence.cpp src/document.cpp)
target_include_directories(repair_core PUBLIC src third_party)
target_link_libraries(repair_core PUBLIC sqlite3)
add_executable(repair_server src/server.cpp src/network.cpp)
target_link_libraries(repair_server PRIVATE repair_core Threads::Threads)
add_library(repair_api src/api_client.cpp)
target_include_directories(repair_api PUBLIC src third_party)
target_link_libraries(repair_api PUBLIC Threads::Threads)
add_library(repair_labels src/labels.cpp third_party/qrcodegen/qrcodegen.cpp)
target_include_directories(repair_labels PUBLIC src third_party third_party/qrcodegen)
target_link_libraries(repair_labels PUBLIC repair_api)
target_link_libraries(repair_server PRIVATE repair_labels)
if(WIN32)
  target_compile_definitions(repair_server PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_server PRIVATE ws2_32 iphlpapi)
  target_compile_definitions(repair_api PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_api PUBLIC ws2_32)
  add_executable(repair_creator WIN32 src/creator_win.cpp src/label_dialog_win.cpp)
  target_compile_definitions(repair_creator PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_creator PRIVATE repair_api repair_labels comctl32 shell32 gdi32 user32 comdlg32 ole32 uuid)
  add_executable(repair_reader WIN32 src/reader_win.cpp)
  target_compile_definitions(repair_reader PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_reader PRIVATE repair_labels comctl32 shell32 gdi32 user32 ole32 uuid)
  add_custom_command(TARGET repair_creator POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory
    "${CMAKE_CURRENT_SOURCE_DIR}/labels" "$<TARGET_FILE_DIR:repair_creator>/labels")
  if(MINGW)
    target_link_options(repair_creator PRIVATE -municode)
    target_link_options(repair_reader PRIVATE -municode)
    foreach(app repair_server repair_creator repair_reader)
      target_link_options(${app} PRIVATE -static)
    endforeach()
  endif()
endif()
if(MSVC)
  target_compile_options(repair_core PRIVATE /utf-8 /W4)
  target_compile_options(repair_server PRIVATE /utf-8 /W4)
  target_compile_options(repair_api PRIVATE /utf-8 /W4)
  target_compile_options(repair_labels PRIVATE /utf-8 /W4)
  if(TARGET repair_creator)
    target_compile_options(repair_creator PRIVATE /utf-8 /W4)
    target_compile_options(repair_reader PRIVATE /utf-8 /W4)
  endif()
else()
  target_compile_options(repair_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_compile_options(repair_server PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_compile_options(repair_api PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(repair_labels PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(repair_api PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
    target_compile_options(repair_labels PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
  endif()
  if(TARGET repair_creator)
    target_compile_options(repair_creator PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(repair_reader PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(repair_creator PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
      target_compile_options(repair_reader PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
    endif()
  endif()
endif()
add_custom_command(TARGET repair_server POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/templates" "$<TARGET_FILE_DIR:repair_server>/templates"
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:repair_server>/web"
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/labels" "$<TARGET_FILE_DIR:repair_server>/labels"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/config.example.json" "$<TARGET_FILE_DIR:repair_server>/config.example.json")
include(CTest)
if(BUILD_TESTING)
  add_executable(core_tests tests/core_tests.cpp)
  target_link_libraries(core_tests PRIVATE repair_core)
  if(MSVC)
    target_compile_options(core_tests PRIVATE /utf-8)
  endif()
  add_test(NAME core_tests COMMAND core_tests)
  add_executable(client_tests tests/client_tests.cpp)
  target_link_libraries(client_tests PRIVATE repair_api)
  if(WIN32)
    target_compile_definitions(client_tests PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  endif()
  if(MSVC)
    target_compile_options(client_tests PRIVATE /utf-8)
  endif()
  add_test(NAME client_tests COMMAND client_tests)
  add_executable(document_tests tests/document_tests.cpp)
  target_link_libraries(document_tests PRIVATE repair_core)
  if(MSVC)
    target_compile_options(document_tests PRIVATE /utf-8)
  endif()
  add_test(NAME document_tests COMMAND document_tests)
  add_executable(labels_tests tests/labels_tests.cpp)
  target_link_libraries(labels_tests PRIVATE repair_labels repair_core)
  if(MSVC)
    target_compile_options(labels_tests PRIVATE /utf-8)
  endif()
  add_test(NAME labels_tests COMMAND labels_tests)
endif()
