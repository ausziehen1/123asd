cmake_minimum_required(VERSION 3.20)
project(RepairCards VERSION 0.5.0 LANGUAGES C CXX)
if(MSVC)
  add_compile_options(/utf-8)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(sqlite3 STATIC third_party/sqlite3.c)
target_include_directories(sqlite3 PUBLIC third_party)
target_compile_definitions(sqlite3 PRIVATE SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
target_link_libraries(sqlite3 PUBLIC Threads::Threads ${CMAKE_DL_LIBS})
add_library(repair_zip STATIC third_party/miniz/miniz.c third_party/miniz/miniz_zip.c third_party/miniz/miniz_tdef.c third_party/miniz/miniz_tinfl.c)
target_include_directories(repair_zip PUBLIC third_party/miniz)
add_library(repair_one_c src/one_c.cpp third_party/pugixml.cpp)
target_include_directories(repair_one_c PUBLIC src third_party)
target_link_libraries(repair_one_c PUBLIC repair_zip)
add_library(repair_core src/store.cpp src/sequence.cpp src/document.cpp)
target_include_directories(repair_core PUBLIC src third_party)
target_link_libraries(repair_core PUBLIC sqlite3 repair_one_c)
add_library(repair_update src/update.cpp)
target_include_directories(repair_update PUBLIC src third_party)
target_link_libraries(repair_update PUBLIC repair_zip)
add_executable(repair_server src/server.cpp src/network.cpp)
target_link_libraries(repair_server PRIVATE repair_core Threads::Threads)
add_library(repair_api src/api_client.cpp)
target_include_directories(repair_api PUBLIC src third_party)
target_link_libraries(repair_api PUBLIC Threads::Threads)
add_library(repair_labels src/labels.cpp third_party/qrcodegen/qrcodegen.cpp)
target_include_directories(repair_labels PUBLIC src third_party third_party/qrcodegen)
target_link_libraries(repair_labels PUBLIC repair_api)
target_link_libraries(repair_server PRIVATE repair_labels)
if(WIN32)
  target_compile_definitions(repair_server PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_server PRIVATE ws2_32 iphlpapi)
  target_compile_definitions(repair_api PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_api PUBLIC ws2_32)
  enable_language(RC)
  configure_file(resources/app.rc.in ${CMAKE_CURRENT_BINARY_DIR}/app.rc @ONLY)
  add_library(repair_client SHARED src/client_dll_win.cpp src/creator_win.cpp src/reader_win.cpp src/label_dialog_win.cpp src/viewer_win.cpp src/ui_theme_win.cpp)
  set_target_properties(repair_client PROPERTIES PREFIX "" OUTPUT_NAME "RepairClient")
  target_include_directories(repair_client SYSTEM PRIVATE third_party/webview2)
  target_compile_definitions(repair_client PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_client PRIVATE repair_labels comctl32 shell32 gdi32 user32 comdlg32 ole32 uuid uxtheme)
  add_executable(repair_updater WIN32 src/updater_win.cpp ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
  set_target_properties(repair_updater PROPERTIES OUTPUT_NAME "Updater")
  target_compile_definitions(repair_updater PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
  target_link_libraries(repair_updater PRIVATE repair_update comdlg32)
  if(MINGW)
    target_link_options(repair_updater PRIVATE -static -municode)
  endif()
  foreach(role creator reader repair)
    add_executable(repair_${role} WIN32 src/launcher_win.cpp ${CMAKE_CURRENT_BINARY_DIR}/app.rc)
    target_compile_definitions(repair_${role} PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
    add_dependencies(repair_${role} repair_client)
    if(role STREQUAL "creator")
      target_compile_definitions(repair_${role} PRIVATE CLIENT_ROLE=0)
      set_target_properties(repair_${role} PROPERTIES OUTPUT_NAME "Creator")
    elseif(role STREQUAL "reader")
      target_compile_definitions(repair_${role} PRIVATE CLIENT_ROLE=1)
      set_target_properties(repair_${role} PROPERTIES OUTPUT_NAME "Reader")
    else()
      target_compile_definitions(repair_${role} PRIVATE CLIENT_ROLE=2)
      set_target_properties(repair_${role} PROPERTIES OUTPUT_NAME "Repair")
    endif()
    if(MINGW)
      target_link_options(repair_${role} PRIVATE -static -municode)
    endif()
  endforeach()
  if(MSVC)
    target_compile_options(repair_client PRIVATE /utf-8 /W4)
  else()
    target_compile_options(repair_client PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(repair_client PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
    endif()
  endif()
  if(MINGW)
    target_link_options(repair_client PRIVATE -static)
    target_link_options(repair_server PRIVATE -static)
  endif()
  add_custom_command(TARGET repair_client POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/installation.json" "$<TARGET_FILE_DIR:repair_client>/installation.json"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/third_party/webview2/WebView2Loader.dll" "$<TARGET_FILE_DIR:repair_client>/WebView2Loader.dll"
    COMMAND ${CMAKE_COMMAND} -E copy_if_different "${CMAKE_CURRENT_SOURCE_DIR}/theme.json" "$<TARGET_FILE_DIR:repair_client>/theme.json"
    COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/labels" "$<TARGET_FILE_DIR:repair_client>/labels")

endif()
if(MSVC)
  target_compile_options(repair_core PRIVATE /utf-8 /W4)
  target_compile_options(repair_server PRIVATE /utf-8 /W4)
  target_compile_options(repair_api PRIVATE /utf-8 /W4)
  target_compile_options(repair_labels PRIVATE /utf-8 /W4)
  if(TARGET repair_creator)
    target_compile_options(repair_creator PRIVATE /utf-8 /W4)
    target_compile_options(repair_reader PRIVATE /utf-8 /W4)
  endif()
else()
  target_compile_options(repair_core PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_compile_options(repair_server PRIVATE -Wall -Wextra -Wpedantic -finput-charset=UTF-8 -fexec-charset=UTF-8)
  target_compile_options(repair_api PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(repair_labels PRIVATE -Wall -Wextra -Wpedantic)
  if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
    target_compile_options(repair_api PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
    target_compile_options(repair_labels PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
  endif()
  if(TARGET repair_creator)
    target_compile_options(repair_creator PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(repair_reader PRIVATE -Wall -Wextra -Wpedantic)
    if(CMAKE_CXX_COMPILER_ID STREQUAL "GNU")
      target_compile_options(repair_creator PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
      target_compile_options(repair_reader PRIVATE -finput-charset=UTF-8 -fexec-charset=UTF-8)
    endif()
  endif()
endif()
add_custom_command(TARGET repair_server POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/templates" "$<TARGET_FILE_DIR:repair_server>/templates"
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/web" "$<TARGET_FILE_DIR:repair_server>/web"
  COMMAND ${CMAKE_COMMAND} -E copy_directory
  "${CMAKE_CURRENT_SOURCE_DIR}/labels" "$<TARGET_FILE_DIR:repair_server>/labels"
  COMMAND ${CMAKE_COMMAND} -E copy_if_different
  "${CMAKE_CURRENT_SOURCE_DIR}/config.example.json" "$<TARGET_FILE_DIR:repair_server>/config.example.json")
